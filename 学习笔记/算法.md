public int findMin(int[] nums) {
        int left = 0;
        int right = nums.length - 1;
        if(nums.length == 1){
            return nums[0];
        }
        if (nums[left] < nums[nums.length - 1]) {
            return nums[left];
        }
        while (left + 1< right) {
            int mid = left + (right - left) / 2;
            if (nums[left] > nums[mid]) {
                right = mid;
            }
            else if (nums[mid] > nums[left]) {
                left = mid;
            }
        }
        return nums[right];
    }

            Arrays.sort(startTime);
            Arrays.sort(endTime);
            int lessStart = upperbound(startTime, 0, startTime.length - 1, queryTime);
            int lessEnd = lowerbound(endTime, 0, endTime.length - 1, queryTime);

        public static int upperbound(int[] arr, int l, int r, int target) {
            int ans = r + 1;
            while (l <= r) {
                int mid = l + ((r - l) >> 1);
                if (arr[mid] > target) {
                    ans = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            return ans;
        }

        public static int lowerbound(int[] arr, int l, int r, int target) {
            int ans = r + 1;
            while (l <= r) {
                int mid = l + ((r - l) >> 1);
                if (arr[mid] >= target) {
                    ans = mid;
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            return ans;
        }

         */
        int left = 0, right = numbers.length - 1;

            int sum = numbers[left] + numbers[right];
            if (sum == target) {
                return new int[]{left + 1, right + 1};

        public int busyStudent(int[] startTime, int[] endTime, int queryTime) {
            int n = startTime.length;
            int maxEndTime = Arrays.stream(endTime).max().getAsInt();
            if (queryTime > maxEndTime) {   //如果检查时间大于完成时间最晚的同学，就没有正在写作业的同学

            }
            int[] cnt = new int[maxEndTime + 2]; 
            for (int i = 0; i < n; i++) {   
                cnt[startTime[i]]++;    //开始时间
                cnt[endTime[i] + 1]--;  //endTime[i] 结束后，学生减一
            }
            int ans = 0;
            for (int i = 0; i <= queryTime; i++) {  //计算学生
                ans += cnt[i];
            }
            return ans;
        }
    }

        public int triangleNumber(int[] nums) {
            int n = nums.length;
            Arrays.sort(nums);  //数组排序
            int ans = 0;
            for (int i = 0; i < n; ++i) {   //固定第一条边
                for (int j = i + 1; j < n; ++j) {   //固定第二条边
                    int left = j + 1, right = n - 1, k = j; //从第二条边之后找到符合第三条边的范围
                    while (left <= right) {     //二分查找找到第三条边
                        int mid = (left + right) / 2;
                        if (nums[mid] < nums[i] + nums[j]) {    //计算两边之和是否大于第三边
                            k = mid;
                            left = mid + 1;
                        } else {
                            right = mid - 1;
                        }
                    }
                    ans += k - j;   //三角形加一
                }
            }
            return ans;
        }
    }

        public List<List<Integer>> subsets(int[] nums) {
            int n = nums.length;
            for (int mask = 0; mask < (1 << n); ++mask) {
                t.clear();
                for (int i = 0; i < n; ++i) {
                    if ((mask & (1 << i)) != 0) {
                        t.add(nums[i]);
                    }
                }
                ans.add(new ArrayList<Integer>(t));
            }
            return ans;
        }

        class Solution {
            List<Integer> t = new ArrayList<Integer>();
            List<List<Integer>> ans = new ArrayList<List<Integer>>();

            public List<List<Integer>> subsets(int[] nums) {
                dfs(0, nums);
                return ans;
            }

            public void dfs(int cur, int[] nums) {      //cur 是当前已经遍历到数组的第几个数
                if (cur == nums.length) {          //达到条件后开始返回，从最长的数组依次添加
                    ans.add(new ArrayList<Integer>(t));
                    return;
                }
                t.add(nums[cur]);       //选择不同的数字加入数组组成一个子集
                dfs(cur + 1, nums);     
                t.remove(t.size() - 1);     //删去前一个加入的值
                dfs(cur + 1, nums);         //再进行选择
            }
        }

    class Solution {
    public int[] getAverages(int[] nums, int k) {
        int n = nums.length;
        int[] avgs = new int[n];
        Arrays.fill(avgs, -1);
        long s = 0; // 维护窗口元素和（注意类型！）
        for (int i = 0; i < n; i++) {
            // 1. 进入窗口
            s += nums[i];
            if (i < k * 2) { // 窗口大小不足 2k+1
                continue;
            }
            // 2. 记录答案
            avgs[i - k] = (int) (s / (k * 2 + 1));
            // 3. 离开窗口
            s -= nums[i - k * 2];
        }
        return avgs;
    }

        class Solution {
            public boolean predictTheWinner(int[] nums) {
                int length = nums.length;
                int[] dp = new int[length];
                for (int i = 0; i < length; i++) {
                    dp[i] = nums[i];
                }
                for (int i = length - 2; i >= 0; i--) {
                    for (int j = i + 1; j < length; j++) {
                        dp[j] = Math.max(nums[i] - dp[j], nums[j] - dp[j - 1]);
                    }
                }
                return dp[length - 1] >= 0;
            }
        }

        class Solution {
            public boolean PredictTheWinner(int[] nums) {
                return total(nums, 0, nums.length - 1, 1) >= 0;
            }

            public int total(int[] nums, int start, int end, int turn) {
                if (start == end) {
                    return nums[start] * turn;
                }
                int scoreStart = nums[start] * turn + total(nums, start + 1, end, -turn);
                int scoreEnd = nums[end] * turn + total(nums, start, end - 1, -turn);
                return Math.max(scoreStart * turn, scoreEnd * turn) * turn;
            }
        }

        public boolean canMeasureWater(int x, int y, int z) {
            Deque<int[]> stack = new LinkedList<int[]>(); //创建一个栈
            stack.push(new int[]{0, 0});    //加入水桶中的数值
            Set<Long> seen = new HashSet<Long>();   //哈希表存储已经获得的值
            while (!stack.isEmpty()) {  //栈不是空的
                if (seen.contains(hash(stack.peek()))) {//检查栈顶元素
                    stack.pop();
                    continue;
                }
                seen.add(hash(stack.peek()));//没有就加入

                int[] state = stack.pop();  //接收值
                int remain_x = state[0], remain_y = state[1];
                if (remain_x == z || remain_y == z || remain_x + remain_y == z) {//如果满足条件返回true
                    return true;
                }
                //遍历所有的可能性，并加入栈中

                // 把 X 壶灌满。
                stack.push(new int[]{x, remain_y});
                // 把 Y 壶灌满。
                stack.push(new int[]{remain_x, y});
                // 把 X 壶倒空。
                stack.push(new int[]{0, remain_y});
                // 把 Y 壶倒空。
                stack.push(new int[]{remain_x, 0});
                // 把 X 壶的水灌进 Y 壶，直至灌满或倒空。
                stack.push(new int[]{remain_x - Math.min(remain_x, y - remain_y), remain_y + Math.min(remain_x, y - remain_y)});
                // 把 Y 壶的水灌进 X 壶，直至灌满或倒空。
                stack.push(new int[]{remain_x + Math.min(remain_y, x - remain_x), remain_y - Math.min(remain_y, x - remain_x)});
            }
            return false;
        }

    public int maxDistance(int[] position, int m) {
        Arrays.sort(position);
        int left = 1, right = position[position.length - 1] - position[0], ans = -1;    //定义变量
        while (left <= right) {     //二分查找找到合适的最小间距
            int mid = (left + right) / 2;
            if (check(mid, position, m)) {  //调用函数
                ans = mid;  //记录结果
                left = mid + 1;     //向右边遍历，间隔变大找值
            } else {
                right = mid - 1;    //向左边遍历，间隔太大，缩小
            }
        }
        return ans;
    }

    public boolean check(int x, int[] position, int m) {
        int pre = position[0], cnt = 1;
        for (int i = 1; i < position.length; ++i) {
            if (position[i] - pre >= x) {   //大于等于最小距离才能通过
                pre = position[i];
                cnt += 1;   //计算放的球数
            }
        }
        return cnt >= m;
    }

public long totalCost(int[] costs, int k, int candidates) {
        int n = costs.length;
        PriorityQueue<int[]> pq = new PriorityQueue<int[]>((a, b) -> a[0] != b[0] ? a[0] - b[0] : a[1] - b[1]);//优先队列记录升序数组代价和索引
        int left = candidates - 1, right = n - candidates;  //计算左右指针起始位置
        if (left + 1 < right) {     //把符合条件的数组值添加到堆中
            for (int i = 0; i <= left; ++i) {
                pq.offer(new int[]{costs[i], i});
            }
            for (int i = right; i < n; ++i) {
                pq.offer(new int[]{costs[i], i});
            }
        } else {
            for (int i = 0; i < n; ++i) {   //全部符合条件
                pq.offer(new int[]{costs[i], i});
            }
        }
        long ans = 0;
        for (int i = 0; i < k; ++i) {
            int[] arr = pq.poll();  //出堆
            int cost = arr[0], id = arr[1]; //
            ans += cost;
            if (left + 1 < right) { //添加下一个符合条件的值
                if (id <= left) {
                    ++left;
                    pq.offer(new int[]{costs[left], left});
                } else {
                    --right;
                    pq.offer(new int[]{costs[right], right});
                }
            }
        }
        return ans;
    }
